    (function initPwaIcon() {
      function drawIcon(size) {
        var canvas = document.createElement("canvas");
        canvas.width = size;
        canvas.height = size;
        var ctx = canvas.getContext("2d");

        var bg = ctx.createLinearGradient(0, 0, size, size);
        bg.addColorStop(0, "#2d805c");
        bg.addColorStop(0.55, "#216347");
        bg.addColorStop(1, "#184733");
        ctx.fillStyle = bg;
        ctx.fillRect(0, 0, size, size);

        var pad = size * 0.07;
        ctx.strokeStyle = "rgba(255, 255, 255, 0.22)";
        ctx.lineWidth = Math.max(2, size * 0.014);
        ctx.beginPath();
        ctx.roundRect(pad, pad, size - pad * 2, size - pad * 2, size * 0.18);
        ctx.stroke();

        var cx = size * 0.5, cy = size * 0.41;
        var boxW = size * 0.25, boxH = size * 0.135, boxD = size * 0.15;
        ctx.strokeStyle = "#ffffff";
        ctx.lineWidth = Math.max(3, size * 0.032);
        ctx.lineJoin = "round";
        ctx.lineCap = "round";

        ctx.beginPath();
        ctx.moveTo(cx, cy - boxH);
        ctx.lineTo(cx + boxW, cy);
        ctx.lineTo(cx + boxW, cy + boxD);
        ctx.lineTo(cx, cy + boxD + boxH);
        ctx.lineTo(cx - boxW, cy + boxD);
        ctx.lineTo(cx - boxW, cy);
        ctx.closePath();
        ctx.fillStyle = "rgba(255, 255, 255, 0.14)";
        ctx.fill();
        ctx.stroke();

        ctx.beginPath();
        ctx.moveTo(cx - boxW, cy);
        ctx.lineTo(cx, cy + boxH);
        ctx.lineTo(cx + boxW, cy);
        ctx.moveTo(cx, cy + boxH);
        ctx.lineTo(cx, cy + boxD + boxH);
        ctx.stroke();

        ctx.fillStyle = "#f2c94c";
        ctx.beginPath();
        ctx.ellipse(cx + boxW * 0.88, cy - boxH * 0.85, size * 0.055, size * 0.03, -Math.PI / 4, 0, Math.PI * 2);
        ctx.fill();

        ctx.fillStyle = "#ffffff";
        ctx.font = "900 " + Math.round(size * 0.19) + "px 'Noto Sans TC', 'PingFang TC', 'Microsoft JhengHei', sans-serif";
        ctx.textAlign = "center";
        ctx.textBaseline = "middle";
        ctx.fillText("新湖", cx, size * 0.77);

        return canvas.toDataURL("image/png");
      }

      try {
        var icon180 = drawIcon(180);
        var icon512 = drawIcon(512);
        document.getElementById("appFavicon").href = icon180;
        document.getElementById("appleTouchIcon").href = icon180;

          ]
        };

        document.getElementById("webManifestLink").href = URL.createObjectURL(blob);
      } catch (e) {}
